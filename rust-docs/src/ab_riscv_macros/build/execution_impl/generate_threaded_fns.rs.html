12/// Calling convention used for functions involved in threaded execution, and the target features
13/// they need.
14///
15/// The default Rust calling convention returns nothing larger than a pointer pair in registers, so
16/// the outcome of execution would come back through memory, costing an argument register in every
17/// handler of the chain. An explicitly pinned convention is what makes the register return
18/// possible through `OpaqueThreadedExecutionResult`, which internally is composed of:
19/// * 256-bit vector register on x86-64 using `sysv64` ABI, which also gives handlers six argument
20///   registers on Windows
21/// * homogeneous aggregates on Aarch64
22///
23/// Everywhere else the outcome comes back through memory, and nothing special is needed for that.
24fn handler_abi() -> anyhow::Result<(Option<Abi>, Option<Attribute>)> {
25    let target_arch = env::var("CARGO_CFG_TARGET_ARCH").context(
26        "Failed to retrieve `CARGO_CFG_TARGET_ARCH` environment variable, make sure to call \
27        `process_instruction_macros` from `build.rs`",
28    )?;
29
30    // TODO: `extern "Rust"` returning such a value through memory where the platform's own
31    //  convention returns it in registers is a bug rather than something inherent, so this
32    //  pinning can go away (except on x86-64 Windows) once it is fixed:
33    //  https://github.com/rust-lang/rust/issues/161381
34    Ok(if target_arch == "x86_64" {
35        (
36            Some(parse_quote! { extern "sysv64" }),
37            // Miri emulates a CPU without AVX and refuses to call a function that requires it, so
38            // unless it is running code built for AVX in the first place, the feature is disabled
39            Some(parse_quote! {
40                #[cfg_attr(
41                    any(not(miri), target_feature = "avx"),
42                    target_feature(enable = "avx")
43                )]
44            }),
45        )
46    } else {
47        (Some(parse_quote! { extern "C" }), None)
48    })
49}

51/// Where clause for the generated threaded items.
52///
53/// Threaded execution is never `const` (dispatch goes through a table of function pointers, which
54/// `const fn` does not allow), so `[const]` bounds are relaxed to ordinary ones, exactly like they
55/// are for a non-`const` execution implementation, and the fetching half of the program counter is
56/// required on top of what `execute()` needs.
57fn threaded_where_clause(generics: &Generics, self_ty: &Type) -> anyhow::Result<WhereClause> {
58    let Some(mut where_clause) = generics.where_clause.clone() else {
59        return Err(anyhow::anyhow!("Missing where clause"));
60    };
61
62    strip_const_where_predicates(&mut where_clause.predicates);
63
64    where_clause.predicates.push(parse_quote! {
65        PC: InstructionFetcher<#self_ty, Memory>
66    });
67    // Applying an `ExecutionResult` is the executor's job rather than the instruction's, so the
68    // handlers need the register file even for an extension whose own instructions never touch it
69    where_clause.predicates.push(parse_quote! {
70        Regs: RegisterFile<Reg>
71    });
72
73    Ok(where_clause)
74}

76/// Generates tail-call-threaded execution alongside the `match`-based `execute()`.
77///
78/// One handler function per instruction variant is generated. A handler destructures its own
79/// instruction (so it reads only the operands that instruction names), calls the per-variant
80/// execution function that
81/// [`generate_variant_fns`](super::generate_variant_fns::generate_variant_fns) already produced,
82/// applies the `ExecutionResult` it returns, and tail-calls the next handler, so the chain never
83/// returns until execution stops.
84pub(super) fn generate_threaded_fns(
85    enum_name: &Ident,
86    self_ty: &Type,
87    generics: &Generics,
88    variants: &[Rc<Variant>],
89    match_arms: &[Arm],
90) -> anyhow::Result<Vec<Item>> {
91    let generic_params = &generics.params;
92    let where_clause = threaded_where_clause(generics, self_ty)?;
93    let (abi, target_feature) = handler_abi()?;
94    let enum_snake_case = enum_name.to_string().to_snake_case();
95    let dispatch_fn_name = format_ident!("dispatch_{enum_snake_case}");
96    let entry_fn_name = format_ident!("execute_{enum_snake_case}_threaded");
97    let dispatch_result_name = format_ident!("{enum_name}ThreadedDispatchResult");
98    let branch_failed_fn_name = format_ident!("{enum_snake_case}_threaded_branch_failed");
99    // What handlers return: the outcome serialized into a shape the target returns in registers
100    let handler_result_ty: Type = parse_quote! {
101        OpaqueThreadedExecutionResult<#self_ty>
102    };
103    // What execution as a whole returns, once the chain is over
104    let result_ty: Type = parse_quote! {
105        ThreadedExecutionResult<#self_ty>
106    };
107
108    let mut generated_items = Vec::with_capacity(variants.len() + 3);
109    let mut dispatch_arms = Vec::with_capacity(variants.len());
110
111    // `FetchInstructionResult` with the handler that executes the fetched instruction attached, and
112    // without the `Continue` variant, which dispatch resolves while fetching rather than passing
113    // on. It never escapes the dispatch step it is created in, so it never materializes.
114    generated_items.push(parse_quote! {
115        enum #dispatch_result_name<I, Handler>
116        where
117            I: Instruction,
118        {
119            Next { instruction: I, handler: Handler },
120            Break,
121            Err(ExecutionError<<I::Reg as Register>::Type>),
122        }
123    });
124
125    // Where a branch whose target `try_set_pc_relative()` refused goes. Kept out of the handlers
126    // and reached with a tail call: none of them is coming back at that point, so nothing they
127    // hold has to survive the call and none of them needs a frame for it. It takes everything a
128    // handler takes, unused arguments included, because `become` requires the two signatures to
129    // match exactly.
130    generated_items.push(parse_quote! {
131        #[expect(
132            clippy::undocumented_unsafe_blocks,
133            reason = "Comments will be stripped, this will suppress some of the lints that are \
134            caused by it"
135        )]
136        #[expect(clippy::allow_attributes, reason = "Attribute below")]
137        #[allow(
138            improper_ctypes_definitions,
139            reason = "Handlers only ever call each other, within this crate"
140        )]
141        #[cold]
142        #[inline(never)]
143        #target_feature
144        unsafe #abi fn #branch_failed_fn_name<#generic_params>(
145            _instruction: #self_ty,
146            mut instruction_fetcher: PC,
147            regs: &mut Regs,
148            env: Env,
149            memory: &mut Memory,
150        ) -> #handler_result_ty
151            #where_clause
152        {
153            // SAFETY: Only reached from a handler whose `try_set_pc_relative()` returned `false`,
154            // with nothing in between having observed the program counter
155            match unsafe { instruction_fetcher.failed_branch(memory) } {
156                Ok(::core::ops::ControlFlow::Continue(())) => {}
157                Ok(::core::ops::ControlFlow::Break(())) => {
158                    // SAFETY: Platform support is checked before the chain is entered
159                    return unsafe {
160                        OpaqueThreadedExecutionResult::new(
161                            ThreadedExecutionResult::stopped(instruction_fetcher.get_pc()),
162                        )
163                    };
164                }
165                Err(error) => {
166                    // SAFETY: Platform support is checked before the chain is entered
167                    return unsafe {
168                        OpaqueThreadedExecutionResult::new(
169                            ThreadedExecutionResult::failed(
170                                instruction_fetcher.get_pc(),
171                                error,
172                            ),
173                        )
174                    };
175                }
176            }
177
178            let (instruction, handler) = match #dispatch_fn_name::<#generic_params>(
179                &mut instruction_fetcher,
180                memory,
181            ) {
182                #dispatch_result_name::Next {
183                    instruction,
184                    handler,
185                } => (instruction, handler),
186                #dispatch_result_name::Break => {
187                    // SAFETY: Platform support is checked before the chain is entered
188                    return unsafe {
189                        OpaqueThreadedExecutionResult::new(
190                            ThreadedExecutionResult::stopped(instruction_fetcher.get_pc()),
191                        )
192                    };
193                }
194                #dispatch_result_name::Err(error) => {
195                    // SAFETY: Platform support is checked before the chain is entered
196                    return unsafe {
197                        OpaqueThreadedExecutionResult::new(
198                            ThreadedExecutionResult::failed(
199                                instruction_fetcher.get_pc(),
200                                error,
201                            ),
202                        )
203                    };
204                }
205            };
206
207            // SAFETY: Every handler carries the same target features this one does, which is what
208            // makes them unsafe to call in the first place
209            unsafe {
210                become handler(
211                    instruction,
212                    instruction_fetcher,
213                    regs,
214                    env,
215                    memory,
216                )
217            }
218        }
219    });
220
221    for (variant, arm) in variants.iter().zip(match_arms) {
222        let variant_ident = &variant.ident;
223        let Pat::Struct(pat_struct) = &arm.pat else {
224            return Err(anyhow::anyhow!(
225                "Expected a struct pattern for instruction variant `{variant_ident}`, found `{}`",
226                arm.pat.to_token_stream()
227            ));
228        };
229        let pat_fields = &pat_struct.fields;
230
231        let variant_fn_name = variant_fn_name(enum_name, variant_ident);
232        let handler_fn_name = format_ident!("{variant_fn_name}_threaded");
233
234        let variant_call_args = pat_fields
235            .iter()
236            .filter_map(|field| match field.pat.as_ref() {
237                Pat::Ident(pat_ident) => Some(&pat_ident.ident),
238                _ => None,
239            });
240
241        generated_items.push(parse_quote! {
242            #[expect(
243                clippy::undocumented_unsafe_blocks,
244                reason = "Comments will be stripped, this will suppress some of the lints that \
245                are caused by it"
246            )]
247            #[expect(clippy::allow_attributes, reason = "Attribute below")]
248            #[allow(
249                improper_ctypes_definitions,
250                reason = "Handlers only ever call each other, within this crate"
251            )]
252            // Make sure each handler starts on a cache line boundary
253            #[rustc_align(64)]
254            #target_feature
255            unsafe #abi fn #handler_fn_name<#generic_params>(
256                instruction: #self_ty,
257                mut instruction_fetcher: PC,
258                regs: &mut Regs,
259                mut env: Env,
260                memory: &mut Memory,
261            ) -> #handler_result_ty
262                #where_clause
263            {
264                let Rs1Rs2Operands { rs1, rs2 } = instruction.get_rs1_rs2_operands();
265                let rs1_value = regs.read(rs1);
266                let rs2_value = regs.read(rs2);
267
268                #[expect(clippy::allow_attributes, reason = "Attribute below")]
269                #[allow(
270                    irrefutable_let_patterns,
271                    reason = "True for extensions with a single instruction"
272                )]
273                let #enum_name::#variant_ident { #pat_fields } = instruction else {
274                    // SAFETY: A handler is only ever reached through the dispatch arm for its own
275                    // variant
276                    unsafe {
277                        ::core::hint::unreachable_unchecked();
278                    }
279                };
280
281                // Dispatch leaves the program counter on the instruction it read, and this is what
282                // moves it past it - by a size the compiler folds to a constant, since the variant
283                // is known here. Doing it during the fetch instead makes the address of the next
284                // instruction depend on decoding the current one, which is a load-to-load
285                // dependency in the middle of every dispatch step.
286                //
287                // SAFETY: Dispatch has just peeked this instruction successfully, and this is the
288                // only place that moves past it
289                unsafe {
290                    instruction_fetcher.advance(Instruction::size(&instruction));
291                }
292
293                let execution_result = #variant_fn_name::<#generic_params>(
294                    #( #variant_call_args, )*
295                    rs1_value,
296                    rs2_value,
297                    regs,
298                    &mut env,
299                    memory,
300                    &mut instruction_fetcher,
301                );
302
303                let control_flow = match execution_result {
304                    ExecutionResult::Continue { rd, value } => {
305                        regs.write(rd, value);
306                        Ok(::core::ops::ControlFlow::Continue(()))
307                    }
308                    ExecutionResult::ContinueNoWrite => {
309                        Ok(::core::ops::ControlFlow::Continue(()))
310                    }
311                    ExecutionResult::Branch { offset } => {
312                        // SAFETY: A refused target goes straight to the continuation below, which
313                        // is what calls `failed_branch()` on it
314                        if unsafe {
315                            instruction_fetcher.try_set_pc_relative(
316                                Instruction::size(&instruction),
317                                offset,
318                            )
319                        } {
320                            Ok(::core::ops::ControlFlow::Continue(()))
321                        } else {
322                            // SAFETY: The continuation carries the same target features this
323                            // handler does, which is what makes it unsafe to call in the first
324                            // place
325                            unsafe {
326                                become #branch_failed_fn_name::<#generic_params>(
327                                    instruction,
328                                    instruction_fetcher,
329                                    regs,
330                                    env,
331                                    memory,
332                                )
333                            }
334                        }
335                    }
336                    ExecutionResult::Jump { target } => {
337                        instruction_fetcher.set_pc(memory, target)
338                    }
339                    ExecutionResult::Break => {
340                        ::core::hint::cold_path();
341                        // SAFETY: Platform support is checked before the chain is entered
342                        return unsafe {
343                            OpaqueThreadedExecutionResult::new(
344                                ThreadedExecutionResult::stopped(instruction_fetcher.get_pc()),
345                            )
346                        };
347                    }
348                    ExecutionResult::Err(error) => {
349                        ::core::hint::cold_path();
350                        // SAFETY: Platform support is checked before the chain is entered
351                        return unsafe {
352                            OpaqueThreadedExecutionResult::new(
353                                ThreadedExecutionResult::failed(
354                                    instruction_fetcher.get_pc(),
355                                    error,
356                                ),
357                            )
358                        };
359                    }
360                };
361
362                match control_flow {
363                    Ok(::core::ops::ControlFlow::Continue(())) => {}
364                    Ok(::core::ops::ControlFlow::Break(())) => {
365                        ::core::hint::cold_path();
366                        // SAFETY: Platform support is checked before the chain is entered
367                        return unsafe {
368                            OpaqueThreadedExecutionResult::new(
369                                ThreadedExecutionResult::stopped(instruction_fetcher.get_pc()),
370                            )
371                        };
372                    }
373                    Err(error) => {
374                        ::core::hint::cold_path();
375                        // SAFETY: Platform support is checked before the chain is entered
376                        return unsafe {
377                            OpaqueThreadedExecutionResult::new(
378                                ThreadedExecutionResult::failed(
379                                    instruction_fetcher.get_pc(),
380                                    error,
381                                ),
382                            )
383                        };
384                    }
385                }
386
387                let (instruction, handler) = match #dispatch_fn_name::<#generic_params>(
388                    &mut instruction_fetcher,
389                    memory,
390                ) {
391                    #dispatch_result_name::Next {
392                        instruction,
393                        handler,
394                    } => (instruction, handler),
395                    #dispatch_result_name::Break => {
396                        ::core::hint::cold_path();
397                        // SAFETY: Platform support is checked before the chain is entered
398                        return unsafe {
399                            OpaqueThreadedExecutionResult::new(
400                                ThreadedExecutionResult::stopped(instruction_fetcher.get_pc()),
401                            )
402                        };
403                    }
404                    #dispatch_result_name::Err(error) => {
405                        ::core::hint::cold_path();
406                        // SAFETY: Platform support is checked before the chain is entered
407                        return unsafe {
408                            OpaqueThreadedExecutionResult::new(
409                                ThreadedExecutionResult::failed(
410                                    instruction_fetcher.get_pc(),
411                                    error,
412                                ),
413                            )
414                        };
415                    }
416                };
417
418                // SAFETY: Every handler carries the same target features this one does, which is
419                // what makes them unsafe to call in the first place
420                unsafe {
421                    become handler(
422                        instruction,
423                        instruction_fetcher,
424                        regs,
425                        env,
426                        memory,
427                    )
428                }
429            }
430        });
431
432        dispatch_arms.push(quote! {
433            #enum_name::#variant_ident { .. } => #handler_fn_name::<#generic_params>,
434        });
435    }
436
437    generated_items.push(parse_quote! {
438        #[expect(
439            clippy::type_complexity,
440            reason = "`become` requires an exact signature match and a type alias cannot capture \
441            the enclosing generics, so the handler type is spelled out here"
442        )]
443        #[expect(clippy::allow_attributes, reason = "Attribute below")]
444        #[allow(
445            improper_ctypes_definitions,
446            reason = "Handlers only ever call each other, within this crate"
447        )]
448        #[inline(always)]
449        fn #dispatch_fn_name<#generic_params>(
450            instruction_fetcher: &mut PC,
451            memory: &Memory,
452        ) -> #dispatch_result_name<
453            #self_ty,
454            unsafe #abi fn(
455                #self_ty,
456                PC,
457                &mut Regs,
458                Env,
459                &mut Memory,
460            ) -> #handler_result_ty,
461        >
462            #where_clause
463        {
464            let instruction = loop {
465                match instruction_fetcher.peek_instruction(memory) {
466                    FetchInstructionResult::Instruction(instruction) => {
467                        break instruction;
468                    }
469                    FetchInstructionResult::Continue => {
470                        ::core::hint::cold_path();
471                    }
472                    FetchInstructionResult::Break => {
473                        ::core::hint::cold_path();
474                        return #dispatch_result_name::Break;
475                    }
476                    FetchInstructionResult::Err(error) => {
477                        ::core::hint::cold_path();
478                        return #dispatch_result_name::Err(error);
479                    }
480                }
481            };
482
483            #[expect(
484                clippy::rest_pattern_accessible_field,
485                reason = "Dispatch selects a handler by variant and never looks at any field"
486            )]
487            let handler = match instruction {
488                #( #dispatch_arms )*
489            };
490
491            #dispatch_result_name::Next { instruction, handler }
492        }
493    });
494
495    // The one ordinary call in the whole chain: everything from here on is reached with `become`
496    // and nothing returns until execution stops.
497    //
498    // This is a separate function rather than the body of the trait method because it is the one
499    // that calls handlers, so it has to carry the same target features they do, and a safe trait
500    // method cannot carry target features.
501    generated_items.push(parse_quote! {
502        #[expect(
503            clippy::undocumented_unsafe_blocks,
504            reason = "Comments will be stripped, this will suppress some of the lints that are \
505            caused by it"
506        )]
507        #[inline]
508        #target_feature
509        unsafe fn #entry_fn_name<#generic_params>(
510            mut instruction_fetcher: PC,
511            regs: &mut Regs,
512            env: Env,
513            memory: &mut Memory,
514        ) -> #result_ty
515            #where_clause
516        {
517            let (instruction, handler) = match #dispatch_fn_name::<#generic_params>(
518                &mut instruction_fetcher,
519                memory,
520            ) {
521                #dispatch_result_name::Next {
522                    instruction,
523                    handler,
524                } => (instruction, handler),
525                #dispatch_result_name::Break => {
526                    ::core::hint::cold_path();
527                    return ThreadedExecutionResult::stopped(
528                        instruction_fetcher.get_pc(),
529                    );
530                }
531                #dispatch_result_name::Err(error) => {
532                    ::core::hint::cold_path();
533                    return ThreadedExecutionResult::failed(
534                        instruction_fetcher.get_pc(),
535                        error,
536                    );
537                }
538            };
539
540            // SAFETY: This function carries the same target features every handler does, which is
541            // what makes them unsafe to call in the first place
542            let outcome = unsafe {
543                handler(
544                    instruction,
545                    instruction_fetcher,
546                    regs,
547                    env,
548                    memory,
549                )
550            };
551
552            outcome.into_result()
553        }
554    });
555
556    generated_items.push(parse_quote! {
557        impl<#generic_params> ThreadedExecutableInstruction<
558            Regs,
559            Env,
560            Memory,
561            PC,
562        > for #self_ty
563            #where_clause
564        {
565            #[expect(
566                clippy::undocumented_unsafe_blocks,
567                reason = "Comments will be stripped, this will suppress some of the lints that \
568                are caused by it"
569            )]
570            #[inline(always)]
571            fn execute_threaded(
572                instruction_fetcher: PC,
573                regs: &mut Regs,
574                env: Env,
575                memory: &mut Memory,
576            ) -> #result_ty {
577                if !OpaqueThreadedExecutionResult::<#self_ty>::platform_supported() {
578                    ::core::hint::cold_path();
579                    return ThreadedExecutionResult::failed(
580                        instruction_fetcher.get_pc(),
581                        ExecutionError::UnsupportedPlatform,
582                    );
583                }
584
585                // SAFETY: Platform support for what handlers need was just checked
586                unsafe {
587                    #entry_fn_name::<#generic_params>(
588                        instruction_fetcher,
589                        regs,
590                        env,
591                        memory,
592                    )
593                }
594            }
595        }
596    });
597
598    Ok(generated_items)
